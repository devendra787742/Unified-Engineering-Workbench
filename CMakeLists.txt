cmake_minimum_required(VERSION 3.16)
project(UnifiedEngineeringWorkbench VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(workbench src/workbench.cpp)
target_link_libraries(workbench PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(workbench PRIVATE /W4 /O2)
else()
  target_compile_options(workbench PRIVATE -Wall -Wextra -Wpedantic -O2)
endif()
enable_testing()
add_executable(workbench_tests tests/test_workbench.cpp)
target_compile_definitions(workbench_tests PRIVATE WORKBENCH_TESTING)
target_link_libraries(workbench_tests PRIVATE Threads::Threads)
add_test(NAME core_tests COMMAND workbench_tests)

if(WIN32)
  add_executable(Launch WIN32 src/launcher_win.cpp)
  target_link_libraries(Launch PRIVATE comdlg32 shell32)
  set_target_properties(Launch PROPERTIES OUTPUT_NAME "Launch")
endif()
